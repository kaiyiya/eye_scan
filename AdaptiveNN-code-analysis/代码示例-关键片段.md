## 1. 策略网络 (Policy Network)

### 动作网络 - 预测注视点位置

```python
class policy_net_patch(nn.Module):
    def forward(self, x, step_index):
        # x: (B, 384, 7, 7) - 特征图
        
        # 深度可分离卷积
        x = self.conv1(x)  # (B, 384, 7, 7)
        x = self.act_layer(x)
        
        # 1x1 卷积降维
        x = self.conv2(x)  # (B, 128, 7, 7)
        x = self.act_layer(x)
        
        # 展平
        x = self.flatten(x)  # (B, 6272)
        
        # 全连接层
        x = self.linear1(x)  # (B, 1024)
        x = self.ln_list[step_index][0](x)  # 每个步骤使用不同的 LayerNorm
        x = self.act_layer(x)
        
        x = self.linear2(x)  # (B, 512)
        x = self.ln_list[step_index][1](x)
        x = self.act_layer(x)
        
        # 输出头
        x = self.output_head(x)  # (B, 2) - [x, y] 坐标
        
        return x
```

## 3. 图像 Patch 提取

### 使用仿射变换提取局部区域

```python
def get_img_patches(self, input_frames, actions, patch_size, image_size):
    batchsize = actions.size(0)
    
    # 计算 patch 在原图中的位置
    patch_coordinate = (actions[:, :2] * (image_size - patch_size))
    # actions[:, :2] 是归一化的 [x, y] ∈ [0, 1]
    # 例如: [0.5, 0.5] * (288 - 112) = [88, 88]
    
    # 计算四个角坐标
    x1 = patch_coordinate[:, 1]
    x2 = patch_coordinate[:, 1] + patch_size
    y1 = patch_coordinate[:, 0]
    y2 = patch_coordinate[:, 0] + patch_size
    
    # 构建仿射变换矩阵 (2×3)
    theta = torch.zeros((batchsize, 2, 3), device=device)
    theta[:, 0, 0] = patch_size / image_size  # x 方向缩放
    theta[:, 1, 1] = patch_size / image_size  # y 方向缩放
    theta[:, 0, 2] = -1 + (x1 + x2) / image_size  # x 方向平移
    theta[:, 1, 2] = -1 + (y1 + y2) / image_size  # y 方向平移
    
    # 创建采样网格
    grid = F.affine_grid(
        theta.float(), 
        torch.Size((batchsize, 1, patch_size, patch_size))
    )
    
    # 使用双三次插值提取 patch
    patches = F.grid_sample(input_frames, grid, mode='bicubic')
    # 输出: (B, 3, 112, 112)
    
    return patches
```

## 4. 特征恢复机制

### 将局部特征融合回全局特征图

```python
# 1. 生成恢复权重
recover_policy_weight = self.policy_net_recover(local_features)
# 输入: (B, 49, 384)
# 输出: (B, 49, 25) - 每个位置预测 5×5 的权重矩阵

# 2. 计算邻居坐标
neighbor_coords_index, outlier_mask = self.window_neighbor_grid(...)
# neighbor_coords_index: (B, 49, 25) - 邻居位置的一维索引
# outlier_mask: (B, 49, 25) - 是否超出边界的掩码

# 3. 应用掩码
recover_policy_weight = recover_policy_weight * outlier_mask

# 4. 构建权重矩阵 (B, 49, 49)
relative_position_weight = torch.zeros((B, 49, 49), device=device)
relative_position_weight.scatter_add_(
    dim=2,
    index=neighbor_coords_index,  # (B, 49, 25)
    src=recover_policy_weight     # (B, 49, 25)
).permute(0, 2, 1)

# 5. 特征融合
updated_features = relative_position_weight @ local_features + updated_features
# (B, 49, 49) @ (B, 49, 384) + (B, 49, 384) = (B, 49, 384)
```

**关键理解**:
- `relative_position_weight[i, j]` 表示局部位置 `i` 对全局位置 `j` 的贡献权重
- 使用矩阵乘法实现加权求和

## 5. PPO 奖励计算

### 基于分类损失减少的奖励

```python
# 获取所有分类输出
_list_all_outputs = [x_glance] + x_focus
# [x_glance, x_focus[0], x_focus[1], x_focus[2], x_focus[3]]

# 计算单步奖励：当前步骤损失 - 下一步骤损失
list_to_compute_reward = [
    (F.cross_entropy(_list_all_outputs[step].detach(), targets, reduction='none')
     - F.cross_entropy(_list_all_outputs[step + 1].detach(), targets, reduction='none')
    ).unsqueeze(-1)
    for step in range(seq_l)
]

mb_rewards = torch.cat(list_to_compute_reward, dim=1)  # (B, seq_l)
```

**奖励含义**:
- 如果 `reward > 0`: 执行动作后损失减小，好动作 ✅
- 如果 `reward < 0`: 执行动作后损失增大，坏动作 ❌

## 6. GAE (Generalized Advantage Estimation)

### 计算优势估计

```python
# 输入
mb_rewards = ...  # (B, seq_l) - 奖励
mb_values = old_state_values  # (B, seq_l) - 状态价值

# 初始化
mb_advs = torch.zeros_like(mb_rewards)  # (B, seq_l)
lastgaelam = torch.zeros((B,), device=device)

# 从后往前计算
for t in reversed(range(seq_l)):
    if t == seq_l - 1:  # 最后一步
        nextnonterminal = 0.0
        nextvalues = 0.0
    else:
        nextnonterminal = 1.0
        nextvalues = mb_values[:, t+1]
    
    # TD 误差
    delta = mb_rewards[:, t] + gamma * nextvalues * nextnonterminal - mb_values[:, t]
    
    # GAE
    mb_advs[:, t] = lastgaelam = delta + gamma * ppo_lam * nextnonterminal * lastgaelam

# 回报 = 优势 + 价值
mb_returns = mb_advs + mb_values
```

**GAE 公式**:
```
δ_t = r_t + γ * V(s_{t+1}) - V(s_t)
A_t = δ_t + (γλ) * δ_{t+1} + (γλ)² * δ_{t+2} + ...
```

## 7. PPO 损失计算

### 裁剪目标函数

```python
# 重新评估旧动作
logprobs, state_values, dist_entropy = model(
    old_states=old_states,
    old_actions=old_actions,
    flag='evaluate_policy_net',
    batch_index=batch_index,
    ppo_std_this_iter=ppo_std_this_iter
)

# 重要性采样比率
ratios = torch.exp(logprobs - old_actions_logprobs[batch_index].detach())
# ratios = π_new(a|s) / π_old(a|s)

# PPO 裁剪目标
surr1 = ratios * advantages[batch_index]
surr2 = torch.clamp(ratios, 1 - eps_clip, 1 + eps_clip) * advantages[batch_index]

# PPO 损失
loss_ppo = (
    -torch.min(surr1, surr2).mean()  # 策略损失（裁剪）
    + mse_loss(state_values, mb_returns[batch_index].detach())  # 价值损失
    - 0.01 * dist_entropy.mean()  # 熵正则化
)
```

**PPO 裁剪**:
- `surr1`: 未裁剪的代理目标
- `surr2`: 裁剪后的代理目标
- 取两者最小值，限制策略更新幅度

## 8. 前向传播主流程

### forward_backbone 核心逻辑

```python
def forward_backbone(self, imgs, seq_l=4, ppo_std_this_iter=None):
    # 1. Glance 阶段
    x = F.interpolate(imgs, size=(112, 112), mode='bicubic')
    global_features = self.glance_net(x, remaining_blocks=4)  # 前8层
    x_glance = self.classify(global_features, step=0)  # 初步分类
    
    # 2. Focus 循环
    updated_features = global_features
    for step in range(seq_l):
        # 2.1 策略网络选择注视点
        actions = self.policy_net_patch(updated_features, step)
        if self.training:
            actions = self.sample_actions(actions, ppo_std_this_iter)
        
        # 2.2 提取 patch
        x_patch = self.get_img_patches(imgs, actions)
        
        # 2.3 Focus Net 处理
        local_features = self.focus_net(x_patch, context=feat_to_reuse)
        
        # 2.4 特征恢复
        updated_features = self.recover_features(
            updated_features, local_features, actions
        )
        
        # 2.5 分类
        x_focus = self.classify(updated_features, step + 1)
    
    return {
        'x_glance': x_glance,
        'x_focus': x_focus,
        'actions': actions,
        ...
    }
```

## 10. 知识蒸馏

### 让中间步骤向最终结果学习

```python
out_teacher = expected_outputs['x_focus'][-1].detach()  # 最终结果作为教师

# Glance 结果向最终结果学习
loss_KD = F.kl_div(
    F.log_softmax(x_glance / kd_temp, dim=1),
    F.softmax(out_teacher / kd_temp, dim=1),
    reduction='batchmean'
) * (kd_temp ** 2)

# 每个中间步骤也向最终结果学习
for x_focus_step in x_focus[:-1]:
    loss_KD += F.kl_div(
        F.log_softmax(x_focus_step / kd_temp, dim=1),
        F.softmax(out_teacher / kd_temp, dim=1),
        reduction='batchmean'
    ) * (kd_temp ** 2)
```

**温度缩放**:
- `kd_temp` 用于软化概率分布
- 温度越高，分布越平滑，更容易学习

## 📝 总结

这些代码片段展示了 AdaptiveNN 的核心实现：

1. **策略网络**: 从特征图预测注视点位置
2. **动作采样**: 训练时添加噪声进行探索
3. **Patch 提取**: 使用仿射变换提取局部区域
4. **特征恢复**: 将局部特征融合回全局特征图
5. **PPO 训练**: 使用奖励和优势估计更新策略
6. **交替训练**: 分别更新不同网络组件
